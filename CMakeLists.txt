cmake_minimum_required(VERSION 3.24)
project(WindowPin VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "WindowPin requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
endif()
add_library(windowpin_core src/windows.cpp)
target_include_directories(windowpin_core PUBLIC src)
target_link_libraries(windowpin_core PUBLIC user32 dwmapi)
add_executable(WindowPin WIN32 src/main.cpp resources/app.rc)
target_link_libraries(WindowPin PRIVATE windowpin_core comctl32 shell32 advapi32 gdi32)
if(MSVC)
  target_link_options(WindowPin PRIVATE /MANIFEST:NO)
endif()
enable_testing()
add_executable(windowpin_tests tests/tests.cpp)
target_link_libraries(windowpin_tests PRIVATE windowpin_core)
add_test(NAME windowpin_tests COMMAND windowpin_tests)
add_executable(windowpin_ui_tests tests/ui_tests.cpp resources/app.rc)
target_link_libraries(windowpin_ui_tests PRIVATE windowpin_core comctl32 shell32 advapi32 gdi32)
if(MSVC)
  target_link_options(windowpin_ui_tests PRIVATE /MANIFEST:NO)
endif()
add_test(NAME windowpin_ui_tests COMMAND windowpin_ui_tests)
